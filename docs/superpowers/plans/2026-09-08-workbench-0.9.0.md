- [ ] **Step 1: Failing tests**

```js
import test from 'node:test';
import assert from 'node:assert/strict';
import { buildTraceabilityMatrix } from '../../server/quality/traceability.js';

test('marks requirement covered when traced testcase and evidence exist', () => {
  const project = {
    requirements: [{ id: 'req_1', title: '登录' }],
    testcases: [{ id: 'tc_1', title: '登录成功', trace: ['req_1'] }],
    evidenceBundles: [{ id: 'ev_1', testRunId: 'run_1', itemRefs: ['tc_1'] }],
    testruns: [{ id: 'run_1', status: 'passed' }],
    gates: [],
  };
  const { rows } = buildTraceabilityMatrix(project);
  const row = rows.find((r) => r.requirementId === 'req_1');
  assert.equal(row.status, 'covered');
  assert.deepEqual(row.testCaseIds, ['tc_1']);
});
```

> 注：若现有 evidence 模型字段名不同，以 `implementation-status` 与代码为准调整断言，但语义保持「需求→用例→证据」。
